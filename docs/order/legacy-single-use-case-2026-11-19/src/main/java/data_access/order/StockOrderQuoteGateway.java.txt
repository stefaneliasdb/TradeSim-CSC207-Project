package data_access.order;

import entity.Stock;
import use_case.order.OrderQuoteGateway;

import java.math.BigDecimal;
import java.util.Objects;
import java.util.Optional;
import java.util.function.Function;

/** Adapts the team's stock lookup to the Order quote boundary. */
public final class StockOrderQuoteGateway implements OrderQuoteGateway {
    private final Function<String, Optional<Stock>> stockLookup;
    private final String cashCurrency;

    public StockOrderQuoteGateway(Function<String, Optional<Stock>> stockLookup, String cashCurrency) {
        this.stockLookup = Objects.requireNonNull(stockLookup);
        if (cashCurrency == null || cashCurrency.isBlank()) {
            throw new IllegalArgumentException("Cash currency is required.");
        }
        this.cashCurrency = cashCurrency.strip();
    }

    @Override
    public Optional<BigDecimal> findPrice(String symbol) {
        return stockLookup.apply(symbol)
                .filter(stock -> cashCurrency.equalsIgnoreCase(stock.getCurrency().strip()))
                .map(Stock::getCurrentPrice);
    }
}
