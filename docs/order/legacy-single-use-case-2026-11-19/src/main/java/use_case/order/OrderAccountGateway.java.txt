package use_case.order;

import entity.Order;

import java.util.Optional;

public interface OrderAccountGateway {
    /** Returns empty if the account does not exist. A missing holding has quantity zero. */
    Optional<OrderAccountSnapshot> findAccount(String userID, String symbol);

    /**
     * Atomically replaces the account's cash and the order's symbol quantity and appends
     * the order to history. If the account no longer exists or either current value differs
     * from the expected snapshot, returns false without writing anything. Compare cash by
     * numeric value, regardless of scale. Other holdings and history entries are preserved.
     */
    boolean commitTrade(Order order, OrderAccountSnapshot expected, OrderAccountSnapshot updated);
}
